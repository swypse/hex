Create `src/game/zoom.ts`:

```ts
export const MIN_ZOOM = 0.5;
export const MAX_ZOOM = 2;

export function clampZoom(zoom: number): number {
  return Math.min(MAX_ZOOM, Math.max(MIN_ZOOM, zoom));
}

export function zoomAroundCursor(
  cursor: { x: number; y: number },
  containerPos: { x: number; y: number },
  scale: number,
  nextScale: number,
): { x: number; y: number } {
  const worldX = (cursor.x - containerPos.x) / scale;
  const worldY = (cursor.y - containerPos.y) / scale;
  return {
    x: cursor.x - worldX * nextScale,
    y: cursor.y - worldY * nextScale,
  };
}

export function clampPan(
  pos: { x: number; y: number },
  mapRadius: number,
  hexSize: number,
  scale: number,
  screenW: number,
  screenH: number,
): { x: number; y: number } {
  const halfW = Math.sqrt(3) * mapRadius * hexSize * scale;
  const halfH = 1.5 * mapRadius * hexSize * scale;
  return {
    x: Math.min(Math.max(pos.x, -(halfW - 1)), screenW + (halfW - 1)),
    y: Math.min(Math.max(pos.y, -(halfH - 1)), screenH + (halfH - 1)),
  };
}
```

Note: the clamp allows `pos` to range so the map's near edge can reach the opposite viewport edge but not past it — i.e. `pos.x - halfW <= screenW` and `pos.x + halfW >= 0`. For `halfW > screenW` (zoomed-in large map), the clamp range `[-(halfW-1), screenW + (halfW-1)]` is valid because `screenW + (halfW-1) > -(halfW-1)`.

- [ ] **Step 2: Add a fit-to-screen method and call it in `startGame`**

Add a private method:

```ts
  private applyFitToScreen(): void {
    if (!this.app || !this.map || !this.mapContainer) return;
    const radius = this.map.radius;
    const mapW = 2 * Math.sqrt(3) * radius * HEX_SIZE;
    const mapH = 2 * (1.5 * radius * HEX_SIZE);
    const fit = Math.min(this.app.screen.width / mapW, this.app.screen.height / mapH) * 0.9;
    this.baseScale = fit;
    this.zoom = 1;
    this.pan = {
      x: this.app.screen.width / 2,
      y: this.app.screen.height / 2,
    };
    this.applyTransform();
  }

  private applyTransform(): void {
    if (!this.mapContainer) return;
    const scale = this.baseScale * this.zoom;
    this.mapContainer.scale.set(scale, scale);
    this.mapContainer.position.set(this.pan.x, this.pan.y);
  }
```

Call `this.applyFitToScreen()` right after the first `this.render()` in `startGame` (after `showPopup` line):

```ts

    this.applyFitToScreen();
```

Note: `renderMap` sets the container position to screen center; `applyFitToScreen` then overrides it with `pan` and sets scale. Since `render()` rebuilds the container, `applyTransform()` must be called after every render. Update the `render()` method (Step 3) to re-apply.

with:

```ts
    this.mapContainer = renderMap(this.app, this.map, this.textures, store.players, selection, this.reachableKeys, this.attackableKeys, HEX_SIZE);
    this.mapContainer.eventMode = 'static';

    this.mapContainer.on('wheel', (e) => {
      if (!this.mapContainer) return;
      const factor = e.deltaY < 0 ? 1.1 : 1 / 1.1;
      const scale = this.baseScale * this.zoom;
      const nextZoom = clampZoom(this.zoom * factor);
      const nextScale = this.baseScale * nextZoom;
      this.pan = zoomAroundCursor({ x: e.global.x, y: e.global.y }, this.pan, scale, nextScale);
      this.zoom = nextZoom;
      this.applyTransform();
    });

    this.mapContainer.on('pointerdown', (e) => {
      this.dragging = true;
      this.dragStart = { x: e.global.x, y: e.global.y };
      this.panStart = { ...this.pan };
    });

    this.mapContainer.on('pointermove', (e) => {
      if (!this.dragging || !this.app || !this.map) return;
      const dx = e.global.x - this.dragStart.x;
      const dy = e.global.y - this.dragStart.y;
      this.pan = clampPan(
        { x: this.panStart.x + dx, y: this.panStart.y + dy },
        this.map.radius,
        HEX_SIZE,
        this.baseScale * this.zoom,
        this.app.screen.width,
        this.app.screen.height,
      );
      this.applyTransform();
    });

    this.mapContainer.on('pointerup', (e) => {
      this.dragging = false;
    });

    this.mapContainer.on('pointertap', (e) => {
      if (!this.mapContainer || this.dragging) return;
      const now = Date.now();
      const local = this.mapContainer.toLocal(e.global);
      const h = pixelToHex(local.x, local.y, HEX_SIZE);
      const tile = tileAt(this.map!, h.q, h.r);
      if (now - this.lastTap < 400 && !tile) {
        this.lastTap = 0;
        this.resetView();
        return;
      }
      this.lastTap = now;
      if (tile) {
        this.handleMapClick(h.q, h.r);
      }
    });

    this.app.stage.addChild(this.mapContainer);
    this.applyTransform();
```

Add a private reset method:

```ts
  private resetView(): void {
    this.zoom = 1;
    if (this.app) {
      this.pan = { x: this.app.screen.width / 2, y: this.app.screen.height / 2 };
    }
    this.applyTransform();
  }
```

Note: `pointertap` fires after `pointerup`; the `dragging` guard prevents selection taps after a drag. The double-click detection uses `lastTap` (400ms window) and requires the tap to hit no tile (`!tile`) to reset.
